//SELF
CREATE TABLE enrollment_applications (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    status ENUM('pending','rejected','accepted') DEFAULT 'pending',
    is_fresher BOOLEAN,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE applicant_personal (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    application_id BIGINT NOT NULL,
    full_name_mm VARCHAR(100) NOT NULL,
    full_name_en VARCHAR(100) NOT NULL,
    gender ENUM('male','female') NOT NULL,
    date_of_birth DATE NOT NULL,
    ethnic VARCHAR(50),
    religion VARCHAR(50),
    FOREIGN KEY (application_id) REFERENCES enrollment_applications(id)
);

CREATE TABLE applicant_nrc (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    application_id BIGINT NOT NULL,
    nrc_status ENUM('pending','available') DEFAULT 'available',
    nrc_type SMALLINT,
    nrc_region SMALLINT,
    nrc_township SMALLINT,
    nrc_number VARCHAR(30),
    national_nrc VARCHAR(50),
    FOREIGN KEY (application_id) REFERENCES enrollment_applications(id),
    FOREIGN KEY (nrc_region) REFERENCES nrc_states(id),
    FOREIGN KEY (nrc_township) REFERENCES nrc_townships(id),
    FOREIGN KEY (nrc_type) REFERENCES nrc_types(id)
);

CREATE TABLE applicant_contact (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    application_id BIGINT NOT NULL,
    phone_number VARCHAR(20),
    usual_phone_number VARCHAR(20),
    email VARCHAR(100),
    birth_place VARCHAR(100),
    usual_address_region SMALLINT,
    usual_address_township SMALLINT,
    full_address TEXT,
    FOREIGN KEY (application_id) REFERENCES enrollment_applications(id),
    FOREIGN KEY (usual_address_region) REFERENCES nrc_states(id),
    FOREIGN KEY (usual_address_township) REFERENCES nrc_townships(id)
);

CREATE TABLE applicant_academic (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    application_id BIGINT NOT NULL,
    academic_year VARCHAR(15),
    uni_entrance_rollno VARCHAR(30) NOT NULL,
    uni_entrance_year VARCHAR(30),
    major_year_id SMALLINT UNSIGNED,
    roll_no VARCHAR(10),
    FOREIGN KEY (major_year_id) REFERENCES major(id) ON DELETE CASCADE,
    FOREIGN KEY (application_id) REFERENCES enrollment_applications(id)
);

//PARENT
CREATE TABLE applicant_parents (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    application_id BIGINT NOT NULL,
    parent_type ENUM('mother','father') NOT NULL,
    full_name_mm VARCHAR(100) NOT NULL,
    full_name_en VARCHAR(100) NOT NULL,
    ethnic VARCHAR(50),
    religion VARCHAR(50),
    occupation VARCHAR(100),
    date_of_birth DATE,
    is_deceased BOOLEAN DEFAULT FALSE,
    date_of_death DATE,
    FOREIGN KEY (application_id) REFERENCES enrollment_applications(id) ON DELETE CASCADE
);

CREATE TABLE applicant_parents_nrc (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    applicant_parent_id BIGINT NOT NULL,
    nrc_status ENUM('pending','available') DEFAULT 'available',
    nrc_type SMALLINT,
    nrc_region SMALLINT,
    nrc_township SMALLINT,
    nrc_number VARCHAR(30),
    national_nrc VARCHAR(50),
    FOREIGN KEY (applicant_parent_id) REFERENCES applicant_parents(id),
    FOREIGN KEY (nrc_region) REFERENCES nrc_states(id),
    FOREIGN KEY (nrc_township) REFERENCES nrc_townships(id),
    FOREIGN KEY (nrc_type) REFERENCES nrc_types(id)
);

CREATE TABLE applicant_parents_contact (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    applicant_parent_id BIGINT NOT NULL,
    phone_number VARCHAR(20),
    usual_address_region SMALLINT,
    usual_address_township SMALLINT,
    full_address TEXT,
    FOREIGN KEY (applicant_parent_id) REFERENCES applicant_parents(id),
    FOREIGN KEY (usual_address_region) REFERENCES nrc_states(id),
    FOREIGN KEY (usual_address_township) REFERENCES nrc_townships(id),
);

CREATE TABLE enrollment_files (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    application_id BIGINT NOT NULL,
    file_type_id INT NOT NULL,
    file_path VARCHAR(255) NOT NULL,
    uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (application_id) REFERENCES enrollment_applications(id) ON DELETE CASCADE,
    FOREIGN KEY (file_type_id) REFERENCES file_types(id)
);

//Major preference
CREATE TABLE enrollment_majors (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    application_id BIGINT NOT NULL,
    major_id SMALLINT UNSIGNED NOT NULL,
    preference_rank TINYINT NOT NULL CHECK (preference_rank BETWEEN 1 AND 5),
    FOREIGN KEY (application_id) REFERENCES enrollment_applications(id) ON DELETE CASCADE,
    FOREIGN KEY (major_id) REFERENCES major(id) ON DELETE CASCADE,
    UNIQUE(application_id, major_id)
);


//Dorm 
CREATE TABLE enrollment_dorm (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    application_id BIGINT NOT NULL,
    dorm_fee_responsible ENUM('mother','father','other'),
    other_name VARCHAR(100),
    other_relationship VARCHAR(50),
    other_address TEXT,
    scholarship_support BOOLEAN DEFAULT FALSE,
    scholarship_type VARCHAR(100),
    another_scholarship VARCHAR(100),
    FOREIGN KEY (application_id) REFERENCES enrollment_applications(id) ON DELETE CASCADE
);

//APPROVAL
CREATE TABLE application_approvals (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    application_id BIGINT NOT NULL,
    approver_role ENUM('student_affair', 'hod', 'dorm_warden', 'finance') NOT NULL,
    status ENUM('pending', 'rejected', 'approved') DEFAULT 'pending',
    approved_at TIMESTAMP NULL,
    remarks TEXT,
    FOREIGN KEY (application_id) REFERENCES enrollment_applications(id)
);

    code VARCHAR(20) NOT NULL UNIQUE,

//NRC
CREATE TABLE `nrc_states` (
  `code` varchar(50) NOT NULL,
  `number_en` varchar(10) NOT NULL,
  `number_mm` varchar(10) NOT NULL,
  `name_en` varchar(50) NOT NULL,
  `name_mm` varchar(50) NOT NULL,
  `id` smallint(5) UNSIGNED NOT NULL
) 

CREATE TABLE `nrc_townships` (
  `code` varchar(50) NOT NULL,
  `short_en` varchar(50) DEFAULT NULL,
  `short_mm` varchar(50) DEFAULT NULL,
  `name_en` varchar(50) NOT NULL,
  `name_mm` varchar(50) NOT NULL,
  `state_code` varchar(10) DEFAULT NULL,
  `state_id` smallint(5) UNSIGNED NOT NULL,
  `id` smallint(5) UNSIGNED NOT NULL
)

CREATE TABLE `nrc_types` (
  `name_en` varchar(50) NOT NULL,
  `name_mm` varchar(50) NOT NULL,
  `id` smallint(5) UNSIGNED NOT NULL
)

//pre_enrollment
CREATE TABLE pre_enrollment (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    gender ENUM('male', 'female', 'other') NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    full_name_mm VARCHAR(255) NOT NULL,
    uni_entrance_rollno VARCHAR(50) NOT NULL,
    uni_entrance_academic_year YEAR NOT NULL,
    nrc VARCHAR(50) NOT NULL UNIQUE,
    pass_certificate VARCHAR(255), -- you can store filename or path
    status ENUM('pending', 'accepted', 'rejected') DEFAULT 'pending',
    informed BOOLEAN, --if the user have been informed about the news
    phone_number VARCHAR(20) NOT NULL,
    viber_phone_number VARCHAR(20)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
